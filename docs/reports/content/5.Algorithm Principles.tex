\section{Cơ sở Lý thuyết \& Nguyên lý Thuật toán}
\label{sec:algorithm_principles}

\subsection{Thuật toán Tìm kiếm Không Trọng số (Unweighted Search)}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Breadth-First Search (BFS)}: Sử dụng hàng đợi FIFO (\texttt{collections.deque}). Khám phá đều theo từng mức độ sâu; đảm bảo tìm được đường đi có \textbf{số chặng ít nhất (\textit{minimum hop count})}.
    \item \textbf{Depth-First Search (DFS)}: Sử dụng ngăn xếp LIFO kết hợp tập \texttt{expanded} để chống lặp chu trình. Khám phá sâu dọc theo từng nhánh đường trước khi quay lui (\textit{backtracking}); không đảm bảo tính tối ưu chi phí.
\end{itemize}

\subsection{Thuật toán Tìm kiếm Có Trọng số (Weighted Search)}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Uniform Cost Search (UCS / Dijkstra)}: Sử dụng Min-Priority Queue (\texttt{heapq}) với khóa sắp xếp là $g(n)$ (chi phí tích lũy từ Start đến $n$). Đảm bảo tìm được đường đi có \textbf{tổng chi phí $C^*$ nhỏ nhất} khi mọi $c(e) \ge 0$.
    \item \textbf{A* Search}: Sử dụng Min-Priority Queue với hàm đánh giá:
    \begin{equation}
        f(n) = g(n) + h_s(n)
    \end{equation}
    với Scenario-aware Cost Lower-Bound Heuristic:
    \begin{equation}
        \rho_s = \min_{e\ \text{mở}} \frac{Cost_s(e)}{D_{\text{Haversine}}(e)},
        \qquad h_s(n) = \rho_s \cdot D_{\text{Haversine}}(n, \text{Goal})
    \end{equation}
    Trong đó cạnh đóng và cạnh có khoảng cách Haversine gần 0 không tham gia tính
    $\rho_s$; scenario được freeze trong một lần search.
\end{itemize}

\subsection{Chứng minh Toán học về Tính Đúng đắn của Heuristic}

\begin{theorem}[Tính Chấp nhận được -- Admissibility]
Với mọi cạnh mở, $Cost_s(e) \ge \rho_s D_{\text{Haversine}}(e)$ theo định nghĩa
của $\rho_s$. Vì vậy $h_s(n) = \rho_s D_{\text{Haversine}}(n, \text{Goal})$ là
lower bound và thỏa mãn $0 \le h_s(n) \le h^*(n)$ khi cost không âm và scenario tĩnh.
\end{theorem}

\begin{proof}
1. Khoảng cách đường cong trắc địa (đường chim bay) $D_{\text{Haversine}}(n, \text{Goal})$ là khoảng cách vật lý ngắn nhất giữa hai điểm trên bề mặt Trái Đất. Do đó, với bất kỳ đường đi thực tế nào qua mạng lưới đường bộ, tổng chiều dài $D^*(n, \text{Goal}) \ge D_{\text{Haversine}}(n, \text{Goal})$.\\
2. Mọi thành phần chi phí trong hàm mục tiêu ($T_{\text{ff}}, P_{\text{traffic}}, R_{\text{flood}}$) và các trọng số $w_i$ đều không âm ($\ge 0$).\\
3. Do đó, theo định nghĩa $\rho_s$, chi phí của mỗi cạnh mở thỏa mãn
$c_s(e) \ge \rho_s \cdot D_{\text{Haversine}}(e)$. Tổng chi phí thực tế tối ưu thỏa mãn:
\begin{equation}
    h^*(n) \ge \rho_s \cdot D^*(n, \text{Goal}) \ge \rho_s \cdot D_{\text{Haversine}}(n, \text{Goal}) = h_s(n)
\end{equation}
$\implies h_s(n)$ luôn là một chặn dưới (\textit{lower bound}) hợp lệ, do đó $h_s(n)$ \textbf{Admissible}.
\end{proof}

\begin{theorem}[Tính Nhất quán / Đơn điệu -- Consistency / Monotonicity]
Với mọi cạnh mở $e = (n, n')$, hàm heuristic thỏa mãn $h_s(n) \le c_s(n, n') + h_s(n')$.
\end{theorem}

\begin{proof}
1. Theo bất đẳng thức tam giác cầu (\textit{Spherical Triangle Inequality}) cho khoảng cách địa lý:
\begin{equation}
    D_{\text{Haversine}}(n, \text{Goal}) \le D_{\text{Haversine}}(n, n') + D_{\text{Haversine}}(n', \text{Goal})
\end{equation}
2. Nhân hai vế với $\rho_s \ge 0$:
\begin{equation}
    \rho_s \cdot D_{\text{Haversine}}(n, \text{Goal}) \le \rho_s \cdot D_{\text{Haversine}}(n, n') + \rho_s \cdot D_{\text{Haversine}}(n', \text{Goal})
\end{equation}
\begin{equation}
    \iff h_s(n) \le \rho_s \cdot D_{\text{Haversine}}(n, n') + h_s(n')
\end{equation}
3. Theo định nghĩa $\rho_s$, $\rho_s \cdot D_{\text{Haversine}}(n, n') \le c_s(n, n')$ trên mọi cạnh mở:
\begin{equation}
    \rho_s \cdot D_{\text{Haversine}}(n, n') \le c_s(n, n')
\end{equation}
4. Kết hợp lại suy ra $h_s(n) \le c_s(n, n') + h_s(n') \quad \forall (n, n') \in E \implies h_s(n)$ \textbf{Consistent}.
\end{proof}

\subsection{Thuật toán Tìm kiếm Mở rộng}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Greedy Best-First Search}: Sử dụng hàng đợi ưu tiên với hàm đánh giá $f(n) = h(n)$. Tốc độ thực thi cực nhanh nhưng không đảm bảo tính tối ưu.
    \item \textbf{Bidirectional Search}: Duyệt đồng thời hai luồng BFS: luồng thuận (\textit{Forward}) từ $\text{Start}$ và luồng nghịch (\textit{Backward}) từ $\text{Goal}$ trên đồ thị chuyển vị. Giảm không gian tìm kiếm từ $O(b^d)$ xuống $O(2 \cdot b^{d/2})$.
\end{itemize}

Xét đồ thị mẫu 4 nút: $A \xrightarrow{c=2, h=5} B \xrightarrow{c=4, h=0} D$ và $A \xrightarrow{c=5, h=3} C \xrightarrow{c=1, h=0} D$ với $\text{Start}=A, \text{Goal}=D$.
