\subsection{Quy trình Xử lý Dữ liệu 4 Tầng (4-Tier Data Pipeline)}
Dự án áp dụng quy trình quản trị dữ liệu 4 tầng nghiêm ngặt nhằm đảm bảo tính toàn vẹn và khả năng tái lập:

\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
    node distance=1.5cm and 2.5cm,
    stage/.style={rectangle, draw=blue!70!black, fill=blue!5, rounded corners=4pt, text width=3.8cm, minimum height=1.6cm, align=center, font=\small},
    arrow/.style={thick, ->, >=stealth, color=blue!80!black}
]
    \node[stage] (raw) {\textbf{data/raw/}\\ \footnotesize (OSM Snapshots, UTraffic, Checksums)};
    \node[stage, right=of raw] (interim) {\textbf{data/interim/}\\ \footnotesize (Clean topology, BBox simplified)};
    \node[stage, below=of raw] (fixtures) {\textbf{data/fixtures/}\\ \footnotesize (toy\_graph, golden unit tests)};
\node[stage, below=of interim] (processed) {\textbf{data/processed/}\\ \footnotesize (Landmark 65, Capacity 3229)};

    \draw[arrow] (raw) -- node[above, font=\footnotesize] {scripts/data/} (interim);
    \draw[arrow] (interim) -- node[right, font=\footnotesize] {Build scripts} (processed);
    \draw[arrow] (raw) -- node[left, font=\footnotesize] {SHA-256} (fixtures);
\end{tikzpicture}
\caption{Quy trình xử lý dữ liệu 4 tầng của FloodRoute HCMC.}
\label{fig:data_pipeline}
\end{figure}

\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{\texttt{data/raw/}}: Chứa dữ liệu gốc bất biến (\texttt{osm\_relation\_19407794.geojson}, snapshot Kaggle UTraffic 8.952 nodes/14.043 edges), kiểm tra toàn vẹn bằng mã SHA-256.
    \item \textbf{\texttt{data/interim/}}: Lưu trữ dữ liệu trung gian sau khi trích lọc theo Bounding Box địa lý.
    \item \textbf{\texttt{data/processed/}}: Các graph release phục vụ chạy ứng dụng và thực nghiệm khoa học.
    \item \textbf{\texttt{data/fixtures/}}: Các đồ thị nhỏ phục vụ unit test và golden validation (\texttt{toy\_graph\_v0.1}).
\end{itemize}

\subsection{Hệ thống Nhãn Dữ liệu Minh bạch (Provenance Transparency)}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{\texttt{SOURCE\_BACKED}}: Dữ liệu có nguồn trích xuất xác thực từ OpenStreetMap hoặc UTraffic.
    \item \textbf{\texttt{DERIVED}}: Dữ liệu tính toán từ công thức/script tái lập (cự ly Haversine, Free-flow time).
    \item \textbf{\texttt{SIMULATED / ASSUMPTION}}: Dữ liệu mô phỏng tham số giao thông (hiển thị rõ trên GUI).
    \item \textbf{\texttt{REVIEW\_REQUIRED}}: Dữ liệu map-matching ngập nước đang chờ thẩm định bổ sung.
\end{itemize}
