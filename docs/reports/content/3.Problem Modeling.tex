\section{Mô hình hóa Bài toán \& Hàm Chi phí}
\label{sec:problem_modeling}

\subsection{Mô hình Không gian Trạng thái (State-Space Formulation)}
Bài toán tìm đường được mô hình hóa toán học dưới dạng bài toán tìm kiếm trên không gian trạng thái:
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Tập trạng thái ($S$)}: Tập hợp tất cả các nút giao thông và địa điểm trọng yếu trên mạng lưới đường bộ ($S = V$).
    \item \textbf{Trạng thái bắt đầu ($s_0$)}: Vị trí xuất phát của shipper ($s_0 = \text{Start} \in V$).
    \item \textbf{Tập hành động ($A(s)$)}: Tập các cạnh có hướng xuất phát từ nút $s$ mà không bị đóng trong kịch bản hiện tại:
    \begin{equation}
        A(s) = \{ (s, v) \in E \mid (s, v) \notin \text{ClosedEdges}(\text{Scenario}) \}
    \end{equation}
    \item \textbf{Mô hình chuyển trạng thái ($\text{Result}(s, a)$)}: Khi thực hiện hành động $a = (s, v)$, trạng thái mới đạt được là $v$.
    \item \textbf{Kiểm tra trạng thái đích ($\text{GoalTest}(s)$)}: Kiểm tra xem $s$ có trùng với điểm đích $\text{Goal}$ hay không ($s = \text{Goal}$).
    \item \textbf{Chi phí từng bước ($c(s, a, s')$)}: Chi phí tổng hợp của cạnh $a = (s, s')$ tính theo mô hình chi phí của kịch bản đang chọn.
\end{itemize}

\subsection{Mô hình Chi phí Tổng hợp 4 Thành phần (Composite Cost Function)}
Chi phí của mỗi cạnh có hướng $e = (u, v)$ được tính toán thông qua hàm tuyến tính có trọng số chuẩn hóa:

\begin{equation}
\label{eq:cost_function}
\boxed{C(e) = w_{\text{dist}} \cdot D(e) + w_{\text{time}} \cdot T_{\text{ff}}(e) + w_{\text{cong}} \cdot P_{\text{traffic}}(e) + w_{\text{flood}} \cdot R_{\text{flood}}(e)}
\end{equation}

Trong đó:
\begin{enumerate}[leftmargin=1.5cm]
    \item \textbf{$D(e)$ -- Khoảng cách (Distance)}: Độ dài vật lý của đoạn đường, quy đổi ra kilômét:
    \begin{equation}
        D(e) = \frac{\text{distance\_m}(e)}{1000} \quad (\text{km})
    \end{equation}
    \item \textbf{$T_{\text{ff}}(e)$ -- Thời gian lưu thông tự do (Free-flow Time)}: Thời gian di chuyển ở điều kiện đường thông thoáng với vận tốc thiết kế:
    \begin{equation}
        T_{\text{ff}}(e) = \text{free\_flow\_time\_min}(e) \quad (\text{phút})
    \end{equation}
    \item \textbf{$P_{\text{traffic}}(e)$ -- Độ trễ do kẹt xe (Traffic Delay Penalty)}:
    \begin{align}
        T_{\text{travel}}(e) &= T_{\text{ff}}(e) \times \text{TrafficMultiplier}(e) \\
        P_{\text{traffic}}(e) &= \max\big(0,\, T_{\text{travel}}(e) - T_{\text{ff}}(e)\big) \quad (\text{phút})
    \end{align}
    \textit{Lưu ý thiết kế}: Việc tách riêng $P_{\text{traffic}}$ khỏi $T_{\text{ff}}$ giúp chi phí không bị tính trùng lặp thời gian di chuyển cơ sở.
    \item \textbf{$R_{\text{flood}}(e)$ -- Chỉ số Rủi ro Ngập nước (Flood Risk)}: Giá trị chuẩn hóa $R_{\text{flood}} \in [0.0, 1.0]$ dựa trên mức độ ngập lịch sử (0: khô ráo, 1: ngập sâu nguy hiểm).
    \item \textbf{Bộ trọng số ($w_{\text{dist}}, w_{\text{time}}, w_{\text{cong}}, w_{\text{flood}}$)}: Thỏa mãn điều kiện chuẩn hóa $w_i \ge 0$ và:
    \begin{equation}
        \sum w_i = w_{\text{dist}} + w_{\text{time}} + w_{\text{cong}} + w_{\text{flood}} = 1.0
    \end{equation}
\end{enumerate}
